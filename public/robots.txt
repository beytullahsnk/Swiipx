# ATTENTION EN MODIFIANT CE FICHIER
#
# Un robot n'applique QU'UN SEUL groupe : le plus specifique qui le nomme. Des
# qu'un groupe « User-agent: NomDuRobot » existe, ce robot IGNORE entierement
# « User-agent: * », y compris ses Disallow.
#
# Les groupes nommes ci-dessous ne portaient qu'un « Allow: / ». Bingbot, GPTBot,
# ClaudeBot et les autres avaient donc le droit d'explorer /cart, /checkout,
# /success et /api/ — exactement l'inverse de ce qu'on voulait. Les six Disallow
# sont desormais recopies dans CHAQUE groupe. Toute exclusion ajoutee ici doit
# l'etre partout, sans exception.

Allow: /
Disallow: /api/
Disallow: /cart

Disallow: /checkout/

Disallow: /success/
